function(surface_command name folder)
  add_executable(${name} "${folder}/${name}.cpp")
  target_include_directories(${name} PRIVATE ${FS_INCLUDE_DIRS} "${CMAKE_BINARY_DIR}/help")
  target_link_libraries(${name} PRIVATE utils)
  set(xml "${CMAKE_SOURCE_DIR}/${folder}/${name}.help.xml")
  if(EXISTS "${xml}")
    set(header "${CMAKE_BINARY_DIR}/help/${name}.help.xml.h")
    add_custom_command(OUTPUT "${header}"
      COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/.ddsurfer/help_header.py" "${xml}" "${header}"
      DEPENDS "${xml}" "${CMAKE_SOURCE_DIR}/.ddsurfer/help_header.py" VERBATIM)
    target_sources(${name} PRIVATE "${header}")
  endif()
endfunction()

foreach(command mris_curvature mris_inflate mris_sphere mris_register mri_label2label mri_surf2surf)
  surface_command(${command} ${command})
endforeach()
